cmake_minimum_required(VERSION 3.24)
project(beam_trainer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(FetchContent)
include(CTest)
set(BEAM_QVAC_SOURCE "" CACHE PATH "Prepared pinned QVAC source with Beam's training patch")
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(feature AVX AVX2 FMA F16C SSE42 BMI2)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature TESTS EXAMPLES TOOLS SERVER COMMON)
  set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT BEAM_QVAC_SOURCE)
  find_package(Git REQUIRED)
  FetchContent_Declare(qvac
    URL https://codeload.github.com/tetherto/qvac-fabric-llm.cpp/tar.gz/2633d8d8cab160906601eeb9e3d29a4a12da88cc
    URL_HASH SHA256=73ada05d8933460aca75e6122a3e4822fd648742a285bde611e17eaebc76e21f
    PATCH_COMMAND ${CMAKE_COMMAND} -E env "GIT_CEILING_DIRECTORIES=<SOURCE_DIR>/.."
      ${GIT_EXECUTABLE} apply "${CMAKE_CURRENT_SOURCE_DIR}/qvac.patch")
  FetchContent_GetProperties(qvac)
  if(NOT qvac_POPULATED)
    FetchContent_Populate(qvac)
  endif()
  set(BEAM_QVAC_SOURCE "${qvac_SOURCE_DIR}")
endif()
add_subdirectory("${BEAM_QVAC_SOURCE}" qvac EXCLUDE_FROM_ALL)
if(WIN32)
  FetchContent_Declare(sqlite URL https://www.sqlite.org/2024/sqlite-amalgamation-3460100.zip
    URL_HASH SHA256=77823cb110929c2bcb0f5d48e4833b5c59a8a6e40cdea3936b99e199dbbe5784)
  FetchContent_MakeAvailable(sqlite)
  add_library(trainer_sqlite STATIC "${sqlite_SOURCE_DIR}/sqlite3.c")
  target_include_directories(trainer_sqlite PUBLIC "${sqlite_SOURCE_DIR}")
  target_compile_definitions(trainer_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_THREADSAFE=1)
else()
  find_package(SQLite3 REQUIRED)
  find_package(OpenSSL REQUIRED)
  add_library(trainer_sqlite INTERFACE)
  target_link_libraries(trainer_sqlite INTERFACE SQLite::SQLite3)
endif()
add_executable(beam-trainer trainer.cpp ../learning/store.cpp)
target_include_directories(beam-trainer PRIVATE ../learning ../engine
  "${BEAM_QVAC_SOURCE}/src" "${BEAM_QVAC_SOURCE}/vendor")
target_link_libraries(beam-trainer PRIVATE llama trainer_sqlite)
if(WIN32)
  target_link_libraries(beam-trainer PRIVATE bcrypt)
  if(MINGW)
    target_link_options(beam-trainer PRIVATE -municode)
  endif()
else()
  target_link_libraries(beam-trainer PRIVATE OpenSSL::Crypto)
endif()
foreach(lib llama ggml ggml-base ggml-cpu ggml-vulkan)
  set_target_properties(${lib} PROPERTIES INSTALL_RPATH "$ORIGIN")
  if(WIN32)
    set_target_properties(${lib} PROPERTIES PREFIX "")
  endif()
  if(MINGW)
    target_link_options(${lib} PRIVATE "-Wl,--exclude-libs,ALL")
  endif()
  add_dependencies(beam-trainer ${lib})
  install(TARGETS ${lib} LIBRARY DESTINATION bin RUNTIME DESTINATION bin)
endforeach()
set_target_properties(beam-trainer PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS beam-trainer RUNTIME DESTINATION bin)
install(FILES "${BEAM_QVAC_SOURCE}/LICENSE" DESTINATION licenses RENAME QVAC-MIT.txt)
if(BUILD_TESTING)
  add_executable(beam-trainer-tests ../../tests/trainer.cpp)
  target_include_directories(beam-trainer-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(beam-trainer-tests PRIVATE ggml)
  add_dependencies(beam-trainer-tests ggml-cpu)
  add_test(NAME trainer-loss COMMAND beam-trainer-tests)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native-resume COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/test_native_trainer.py")
  set_tests_properties(native-resume PROPERTIES TIMEOUT 600 ENVIRONMENT
    "BEAM_TEST_NATIVE_TRAINER=$<TARGET_FILE:beam-trainer>;PYTHONPATH=${BEAM_QVAC_SOURCE}/gguf-py")
endif()
