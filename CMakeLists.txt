cmake_minimum_required(VERSION 3.24)
project(beam_ime VERSION 0.1.0 LANGUAGES C CXX)
set(BEAM_VERSION "0.1.0-beta.2")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(GNUInstallDirs)
include(CTest)
include(FetchContent)

option(BEAM_BUILD_DAEMON "Build the model service" ON)
option(BEAM_BUILD_RIME_PLUGIN "Build the system librime plugin" ON)
option(BEAM_VULKAN "Build the optional Vulkan backend" ON)
option(BEAM_NATIVE "Optimize for the build machine (not for distribution)" OFF)
set(BEAM_LLAMA_SOURCE "" CACHE PATH "Existing pinned llama.cpp checkout")
set(BEAM_RIME_SOURCE "" CACHE PATH "librime source matching the system library")
set(BEAM_LLAMA_REVISION "05af0d2b1398394cfa67e1918fee7feabccaa9bc")
if(WIN32)
  set(BEAM_BUILD_RIME_PLUGIN OFF)
endif()
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  find_path(BEAM_JSON_INCLUDE nlohmann/json.hpp)
  if(BEAM_JSON_INCLUDE)
    add_library(beam_json INTERFACE)
    target_include_directories(beam_json INTERFACE "${BEAM_JSON_INCLUDE}")
    add_library(nlohmann_json::nlohmann_json ALIAS beam_json)
  else()
    FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
      GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
    FetchContent_MakeAvailable(json)
  endif()
endif()
add_library(beam_net STATIC src/engine/net.cpp)
target_include_directories(beam_net PUBLIC src/engine)
target_link_libraries(beam_net PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(beam_net PUBLIC ws2_32 bcrypt)
endif()
add_library(beam_client STATIC src/rime/beam_client.cc)
target_include_directories(beam_client PUBLIC src/rime)
target_link_libraries(beam_client PUBLIC beam_net)
add_compile_definitions(BEAM_VERSION="${BEAM_VERSION}")

if(BEAM_BUILD_DAEMON)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_NATIVE ${BEAM_NATIVE} CACHE BOOL "" FORCE)
  set(GGML_VULKAN ${BEAM_VULKAN} CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
  set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
  set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
  foreach(feature TESTS EXAMPLES TOOLS SERVER COMMON)
    set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
  if(BEAM_LLAMA_SOURCE)
    add_subdirectory("${BEAM_LLAMA_SOURCE}" llama.cpp EXCLUDE_FROM_ALL)
  else()
    FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
      GIT_TAG ${BEAM_LLAMA_REVISION})
    FetchContent_GetProperties(llama_cpp)
    if(NOT llama_cpp_POPULATED)
      FetchContent_Populate(llama_cpp)
    endif()
    add_subdirectory("${llama_cpp_SOURCE_DIR}" "${llama_cpp_BINARY_DIR}" EXCLUDE_FROM_ALL)
  endif()
  add_library(beam_engine STATIC src/engine/engine.cpp)
  target_include_directories(beam_engine PUBLIC src/engine)
  target_link_libraries(beam_engine PUBLIC llama)
  add_executable(beamd src/daemon/beamd.cpp)
  target_link_libraries(beamd PRIVATE beam_engine beam_net)
  target_compile_definitions(beamd PRIVATE BEAM_INSTALL_LIBDIR="${CMAKE_INSTALL_LIBDIR}")
  foreach(backend ggml-cpu ggml-vulkan)
    if(TARGET ${backend})
      add_dependencies(beamd ${backend})
    endif()
  endforeach()
  set_target_properties(beamd PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}/beam-ime")
  if(MINGW)
    target_link_options(beamd PRIVATE -municode)
  endif()
  install(TARGETS beamd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  foreach(lib llama ggml ggml-base ggml-cpu ggml-vulkan)
    if(TARGET ${lib})
      set_target_properties(${lib} PROPERTIES INSTALL_RPATH "$ORIGIN")
      if(WIN32)
        set_target_properties(${lib} PROPERTIES PREFIX "")
      endif()
      if(MINGW)
        # Do not re-export the static GCC runtime from each backend DLL.
        target_link_options(${lib} PRIVATE "-Wl,--exclude-libs,ALL")
      endif()
      install(TARGETS ${lib} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/beam-ime
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
  endforeach()
endif()

if(BEAM_BUILD_RIME_PLUGIN)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RIME REQUIRED IMPORTED_TARGET rime)
  pkg_get_variable(RIME_LIBRARY_DIR rime libdir)
  set(BEAM_RIME_PLUGIN_DIR "${RIME_LIBRARY_DIR}/rime-plugins" CACHE PATH "System librime plugin directory")
  if(NOT BEAM_RIME_SOURCE)
    FetchContent_Declare(librime GIT_REPOSITORY https://github.com/rime/librime.git
      GIT_TAG ${RIME_VERSION} GIT_SHALLOW TRUE)
    FetchContent_GetProperties(librime)
    if(NOT librime_POPULATED)
      FetchContent_Populate(librime)
    endif()
    set(BEAM_RIME_SOURCE "${librime_SOURCE_DIR}")
  endif()
  set(RIME_ENABLE_LOGGING 1)
  configure_file("${BEAM_RIME_SOURCE}/src/rime/build_config.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/rime-include/rime/build_config.h")
  find_package(Boost REQUIRED)
  add_library(rime_beam MODULE src/rime/beam_plugin.cc)
  target_include_directories(rime_beam PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/rime-include"
    "${BEAM_RIME_SOURCE}/src" ${Boost_INCLUDE_DIRS})
  target_compile_definitions(rime_beam PRIVATE GLOG_USE_GLOG_EXPORT)
  target_link_libraries(rime_beam PRIVATE beam_client PkgConfig::RIME)
  set_target_properties(rime_beam PROPERTIES PREFIX "" OUTPUT_NAME "librime-beam")
  install(TARGETS rime_beam LIBRARY DESTINATION "${BEAM_RIME_PLUGIN_DIR}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rime-abi.txt" "${RIME_VERSION}\n")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/rime-abi.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/beam-ime)
  add_executable(rime_smoke EXCLUDE_FROM_ALL tests/rime_smoke.cc)
  target_link_libraries(rime_smoke PRIVATE PkgConfig::RIME)
endif()
install(FILES src/rime/beam_ice.schema.yaml DESTINATION ${CMAKE_INSTALL_DATADIR}/beam-ime/rime)
install(DIRECTORY models/ DESTINATION ${CMAKE_INSTALL_DATADIR}/beam-ime/models FILES_MATCHING PATTERN "*.json")
install(PROGRAMS tools/beamctl.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME beamctl)
install(DIRECTORY tools/beamlib DESTINATION ${CMAKE_INSTALL_DATADIR}/beam-ime
  FILES_MATCHING PATTERN "*.py")
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/beam-ime)
install(FILES LICENSE NOTICE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/beam-ime)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(beam_unit_tests tests/unit.cpp)
  target_link_libraries(beam_unit_tests PRIVATE beam_client)
  add_test(NAME core COMMAND beam_unit_tests)
  add_test(NAME python COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p "test_*.py")
  set_tests_properties(python PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/tools:${CMAKE_CURRENT_SOURCE_DIR}/tools/models")
endif()
