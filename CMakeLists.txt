if(BEAM_BUILD_DAEMON)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_NATIVE ${BEAM_NATIVE} CACHE BOOL "" FORCE)
  set(GGML_VULKAN ${BEAM_VULKAN} CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  if(APPLE)
    # Metal does the work; Apple clang already targets apple-m1 for the CPU fallback.
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
  endif()
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
  set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
  set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
  foreach(feature TESTS EXAMPLES TOOLS SERVER COMMON)
    set(LLAMA_BUILD_${feature} OFF CACHE BOOL "" FORCE)
  endforeach()
  if(BEAM_LLAMA_SOURCE)
    set(beam_llama_dir "${BEAM_LLAMA_SOURCE}")
    set(beam_llama_binary_dir llama.cpp)
  else()
    FetchContent_Declare(llama_cpp GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
      GIT_TAG ${BEAM_LLAMA_REVISION})
    FetchContent_GetProperties(llama_cpp)
    if(NOT llama_cpp_POPULATED)
      FetchContent_Populate(llama_cpp)
    endif()
    set(beam_llama_dir "${llama_cpp_SOURCE_DIR}")
    set(beam_llama_binary_dir "${llama_cpp_BINARY_DIR}")
  endif()
  # One public accessor (llama_model_get_tensor) so the engine can fold the translation LoRA into
  # the weights; the patch is applied once, whether the checkout is fetched or supplied.
  find_package(Git REQUIRED)
  set(beam_llama_patch "${CMAKE_CURRENT_SOURCE_DIR}/src/engine/llama-model-tensor.patch")
  set(beam_git ${CMAKE_COMMAND} -E env "GIT_CEILING_DIRECTORIES=${beam_llama_dir}/.." ${GIT_EXECUTABLE})
  execute_process(COMMAND ${beam_git} apply --check --reverse "${beam_llama_patch}"
    WORKING_DIRECTORY "${beam_llama_dir}" RESULT_VARIABLE beam_llama_patched OUTPUT_QUIET ERROR_QUIET)
  if(NOT beam_llama_patched EQUAL 0)
    execute_process(COMMAND ${beam_git} apply "${beam_llama_patch}"
      WORKING_DIRECTORY "${beam_llama_dir}" RESULT_VARIABLE beam_llama_failed)
    if(NOT beam_llama_failed EQUAL 0)
      message(FATAL_ERROR "cannot apply ${beam_llama_patch} to ${beam_llama_dir}")
    endif()
  endif()
  add_subdirectory("${beam_llama_dir}" "${beam_llama_binary_dir}" EXCLUDE_FROM_ALL)
  add_library(beam_engine STATIC src/engine/engine.cpp src/engine/lora_merge.cpp)
  target_include_directories(beam_engine PUBLIC src/engine)
  target_link_libraries(beam_engine PUBLIC llama beam_learning)
  add_executable(beamd src/daemon/beamd.cpp src/learning/runtime.cpp)
  target_link_libraries(beamd PRIVATE beam_engine beam_net beam_learning)
  if(NOT WIN32 AND NOT APPLE)  # macOS hashes with CommonCrypto from libSystem
    find_package(OpenSSL REQUIRED)
    target_link_libraries(beamd PRIVATE OpenSSL::Crypto)
  endif()
  target_compile_definitions(beamd PRIVATE BEAM_BUILD_DATA_DIR="${CMAKE_BINARY_DIR}/data")
  target_compile_definitions(beamd PRIVATE BEAM_INSTALL_LIBDIR="${CMAKE_INSTALL_LIBDIR}")
  foreach(backend ggml-cpu ggml-vulkan ggml-metal)
    if(TARGET ${backend})
      add_dependencies(beamd ${backend})
    endif()
  endforeach()
  if(APPLE)
    set(beam_origin "@loader_path")
  else()
    set(beam_origin "$ORIGIN")
  endif()
  set_target_properties(beamd PROPERTIES INSTALL_RPATH "${beam_origin}/../${CMAKE_INSTALL_LIBDIR}/beam-ime")
  if(MINGW)
    target_link_options(beamd PRIVATE -municode)
  endif()
  install(TARGETS beamd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  foreach(lib llama ggml ggml-base ggml-cpu ggml-vulkan ggml-metal)
    if(TARGET ${lib})
      set_target_properties(${lib} PROPERTIES INSTALL_RPATH "${beam_origin}")
      if(WIN32)
        set_target_properties(${lib} PROPERTIES PREFIX "")
      endif()
      if(MINGW)
        # Do not re-export the static GCC runtime from each backend DLL.
        target_link_options(${lib} PRIVATE "-Wl,--exclude-libs,ALL")
      endif()
      install(TARGETS ${lib} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/beam-ime
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
  endforeach()
endif()

if(BEAM_BUILD_RIME_PLUGIN)
  if(BEAM_RIME_LIBRARY)
    # A prebuilt librime (Squirrel's): the caller supplies its exact source tree and version.
    if(NOT BEAM_RIME_SOURCE OR NOT RIME_VERSION)
      message(FATAL_ERROR "BEAM_RIME_LIBRARY needs BEAM_RIME_SOURCE and RIME_VERSION")
    endif()
    add_library(beam_rime SHARED IMPORTED)
    set_target_properties(beam_rime PROPERTIES IMPORTED_LOCATION "${BEAM_RIME_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${BEAM_RIME_SOURCE}/src")
    get_filename_component(RIME_LIBRARY_DIR "${BEAM_RIME_LIBRARY}" DIRECTORY)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(RIME REQUIRED IMPORTED_TARGET rime)
    pkg_get_variable(RIME_LIBRARY_DIR rime libdir)
    add_library(beam_rime INTERFACE)
    target_link_libraries(beam_rime INTERFACE PkgConfig::RIME)
  endif()
  set(BEAM_RIME_PLUGIN_DIR "${RIME_LIBRARY_DIR}/rime-plugins" CACHE PATH "System librime plugin directory")
  if(NOT BEAM_RIME_SOURCE)
    FetchContent_Declare(librime GIT_REPOSITORY https://github.com/rime/librime.git
      GIT_TAG ${RIME_VERSION} GIT_SHALLOW TRUE)
    FetchContent_GetProperties(librime)
    if(NOT librime_POPULATED)
      FetchContent_Populate(librime)
    endif()
    set(BEAM_RIME_SOURCE "${librime_SOURCE_DIR}")
  endif()
  # Squirrel's librime links glog privately (no glog symbols exported), so the plugin is built
  # without logging; it does not log through glog anyway.
  if(NOT BEAM_RIME_LIBRARY)
    set(RIME_ENABLE_LOGGING 1)
  endif()
  configure_file("${BEAM_RIME_SOURCE}/src/rime/build_config.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/rime-include/rime/build_config.h")
  if(BEAM_BOOST_INCLUDE)
    set(Boost_INCLUDE_DIRS "${BEAM_BOOST_INCLUDE}")
  else()
    find_package(Boost REQUIRED)
  endif()
  add_library(rime_beam MODULE src/rime/beam_plugin.cc)
  target_include_directories(rime_beam PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/rime-include"
    "${BEAM_RIME_SOURCE}/src" ${Boost_INCLUDE_DIRS})
  if(RIME_ENABLE_LOGGING)
    target_compile_definitions(rime_beam PRIVATE GLOG_USE_GLOG_EXPORT)
  endif()
  target_link_libraries(rime_beam PRIVATE beam_client beam_rime)
  # librime loads plugins named librime-<module> with the platform's shared-library suffix.
  set_target_properties(rime_beam PROPERTIES PREFIX "" OUTPUT_NAME "librime-beam")
  if(APPLE)
    set_target_properties(rime_beam PROPERTIES SUFFIX ".dylib")
  endif()
  install(TARGETS rime_beam LIBRARY DESTINATION "${BEAM_RIME_PLUGIN_DIR}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rime-abi.txt" "${RIME_VERSION}\n")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/rime-abi.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/beam-ime)
  add_executable(rime_smoke EXCLUDE_FROM_ALL tests/rime_smoke.cc)
  target_link_libraries(rime_smoke PRIVATE beam_rime)
  if(APPLE)
    # The test copies rime_smoke next to a librime.1.dylib with the plugin in rime-plugins/;
    # resolve librime there only, not in the build machine's Squirrel.app.
    set_target_properties(rime_smoke PROPERTIES SKIP_BUILD_RPATH ON)
    target_link_options(rime_smoke PRIVATE "LINKER:-rpath,@executable_path")
  endif()
endif()
